# แผน 2: db-schema (SQLite + PostgreSQL) + migration + seed — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** สร้าง `packages/db-schema` ที่มี Drizzle schema ครบทุกตารางใน spec §3 สำหรับ 2 dialect (SQLite ฝั่งเครื่อง, PostgreSQL ฝั่งเซิร์ฟเวอร์) พร้อมเทสต์ตรวจว่าสองชุดตรงกัน, migration ที่ generate แล้ว, และตัว seed ที่โหลด `dayo-seed.json` จากแผน 1 ลงฐานข้อมูลทั้งสองแบบได้

**Architecture:** schema เขียนสองชุดด้วยมือ (`src/sqlite/*`, `src/pg/*`) ใช้ helper คอลัมน์ต่อ dialect · เทสต์ parity เทียบชื่อตาราง/คอลัมน์/NOT NULL/PK/FK/UNIQUE ผ่าน `getTableConfig` ของ Drizzle · ตาราง R (อ้างอิง) ฝั่ง pg มี `server_seq` เพิ่มสำหรับ pull cursor · ตาราง T (ธุรกรรม) ฝั่ง pg มี `server_received_at` · ตาราง L (เฉพาะเครื่อง) มีแค่ sqlite · ตาราง server-only มีแค่ pg · seed แปลง code → UUIDv7 แล้ว insert ตามลำดับ FK ใน transaction เดียว · เทสต์ใช้ better-sqlite3 (in-memory) และ PGlite (Postgres ใน WASM) จึงไม่ต้องมีเซิร์ฟเวอร์

**Tech Stack:** drizzle-orm ≥ 0.45 · drizzle-kit ≥ 0.31 · better-sqlite3 ≥ 13 · @electric-sql/pglite (ล่าสุด) · uuid ≥ 14 (`v7`) · Vitest 5

**Spec:** [../specs/2026-09-17-pos-design.md](../specs/2026-09-17-pos-design.md) §3 ทั้งหมด, §6.1 (outbox/sync_state), §7 (idempotency_record, invariant_run), §9 (นำเข้า) · ต่อจาก [แผน 1](2026-09-17-01-foundation-domain.md) (ต้องเสร็จก่อน)

## Global Constraints

- ชนิดคอลัมน์ที่อนุญาต (เพื่อให้ parity ตรวจได้): `text` · `int` (pg `integer`) · `big` (pg `bigint` mode number ↔ sqlite `integer`) · `bool` (pg `boolean` ↔ sqlite `integer` mode boolean) · `json` (pg `jsonb` ↔ sqlite `text` mode json) — **ห้าม** ใช้ timestamp/date/real/numeric: เวลาเก็บเป็น ISO-8601 UTC text, วันทำการเป็น `YYYY-MM-DD` text, ทศนิยมไม่มี (ใช้ milli/satang/usat/bp)
- ชื่อคอลัมน์ใน DB เป็น snake_case, key ใน TS เป็น camelCase · PK ทุกตารางชื่อ `id` text (UUIDv7) ยกเว้นตารางที่ระบุ
- คอลัมน์ `_usat` ใช้ `big` · `_satang`, `_milli`, `_bp`, `qty`, `seq` ใช้ `int`
- **ตารางที่มีเฉพาะ dialect เดียว** ถูก allowlist ในเทสต์ parity: sqlite-only `outbox`, `sync_state` · pg-only `idempotency_record`, `invariant_run` · คอลัมน์ pg-only `server_seq`, `server_received_at`
- ห้ามแก้ schema โดยไม่ generate migration ใหม่ (`pnpm --filter @dayo/db-schema generate`) และเทสต์ migration ต้องผ่านทั้ง sqlite และ pg

- [ ] **Step 2: เขียนเทสต์ parity ให้ตก**

`packages/db-schema/test/parity.test.ts`
```ts
import { describe, expect, it } from 'vitest'
import { getTableConfig as pgConfig, PgTable } from 'drizzle-orm/pg-core'
import { getTableConfig as sqliteConfig, SQLiteTable } from 'drizzle-orm/sqlite-core'
import * as pg from '../src/pg/index.js'
import * as sqlite from '../src/sqlite/index.js'

const SQLITE_ONLY = new Set(['outbox', 'sync_state'])
const PG_ONLY = new Set(['idempotency_record', 'invariant_run'])
const PG_ONLY_COLUMNS = new Set(['server_seq', 'server_received_at'])

type Cat = 'text' | 'int' | 'big' | 'bool' | 'json'
function category(columnType: string): Cat {
  switch (columnType) {
    case 'PgText': case 'SQLiteText': return 'text'
    case 'PgInteger': case 'SQLiteInteger': return 'int'
    case 'PgBigInt53': case 'PgBigSerial53': return 'big'
    case 'PgBoolean': case 'SQLiteBoolean': return 'bool'
    case 'PgJsonb': case 'SQLiteTextJson': return 'json'
    default: throw new Error(`unmapped column type ${columnType}`)
  }
}

type Shape = {
  columns: Record<string, { cat: Cat; notNull: boolean; primary: boolean }>
  fks: string[]      // "col->table.col"
  uniques: string[]  // "colA,colB"
}

function shapeOfPg(t: PgTable): Shape {
  const c = pgConfig(t)
  const columns: Shape['columns'] = {}
  for (const col of c.columns) {
    if (PG_ONLY_COLUMNS.has(col.name)) continue
    // sqlite 'big' is plain integer, so compare big as int
    const cat = category(col.columnType)
    columns[col.name] = { cat: cat === 'big' ? 'int' : cat, notNull: col.notNull, primary: col.primary }
  }
  const fks = c.foreignKeys.map((fk) => { const r = fk.reference(); return `${r.columns.map((x) => x.name).join(',')}->${pgConfig(r.foreignTable as PgTable).name}.${r.foreignColumns.map((x) => x.name).join(',')}` }).sort()
  const uniques = c.uniqueConstraints.map((u) => u.columns.map((x) => x.name).join(',')).sort()
  return { columns, fks, uniques }
}

function shapeOfSqlite(t: SQLiteTable): Shape {
  const c = sqliteConfig(t)
  const columns: Shape['columns'] = {}
  for (const col of c.columns) {
    const cat = category(col.columnType)
    columns[col.name] = { cat: cat === 'big' ? 'int' : cat, notNull: col.notNull, primary: col.primary }
  }
  const fks = c.foreignKeys.map((fk) => { const r = fk.reference(); return `${r.columns.map((x) => x.name).join(',')}->${sqliteConfig(r.foreignTable as SQLiteTable).name}.${r.foreignColumns.map((x) => x.name).join(',')}` }).sort()
  const uniques = c.uniqueConstraints.map((u) => u.columns.map((x) => x.name).join(',')).sort()
  return { columns, fks, uniques }
}

function tables<T>(mod: Record<string, unknown>, guard: (v: unknown) => v is T, nameOf: (t: T) => string): Map<string, T> {
  const out = new Map<string, T>()
  for (const v of Object.values(mod)) if (guard(v)) out.set(nameOf(v), v)
  return out
}

const pgTables = tables(pg, (v): v is PgTable => v instanceof PgTable, (t) => pgConfig(t).name)
const sqliteTables = tables(sqlite, (v): v is SQLiteTable => v instanceof SQLiteTable, (t) => sqliteConfig(t).name)

describe('schema parity sqlite ↔ pg', () => {
  it('has the same set of shared tables', () => {
    const a = [...pgTables.keys()].filter((n) => !PG_ONLY.has(n)).sort()
    const b = [...sqliteTables.keys()].filter((n) => !SQLITE_ONLY.has(n)).sort()
    expect(a).toEqual(b)
    expect(a.length).toBeGreaterThanOrEqual(18)
  })
  for (const name of [...sqliteTables.keys()].filter((n) => !SQLITE_ONLY.has(n))) {
    it(`table ${name} matches`, () => {
      const p = pgTables.get(name)
      expect(p, `pg table ${name} missing`).toBeDefined()
      expect(shapeOfSqlite(sqliteTables.get(name)!)).toEqual(shapeOfPg(p!))
    })
  }
  it('every pg reference table has server_seq and every transaction table has server_received_at', () => {
    const refTables = ['user', 'device', 'setting', 'category', 'product', 'size', 'product_variant', 'sweetness_level', 'channel', 'price', 'recipe', 'recipe_line', 'item', 'purchase_unit', 'bom', 'bom_line', 'equipment', 'customer']
    for (const n of refTables) {
      const cols = pgConfig(pgTables.get(n)!).columns.map((c) => c.name)
      expect(cols, n).toContain('server_seq')
    }
  })
})
```

```ts
import type { Seed } from '@dayo/contracts'
import type * as s from '../sqlite/index.js'

type Ins<T extends { $inferInsert: unknown }> = T['$inferInsert']

export type SeedRows = {
  categories: Ins<typeof s.category>[]
  items: Ins<typeof s.item>[]
  purchaseUnits: Ins<typeof s.purchaseUnit>[]
  boms: Ins<typeof s.bom>[]
  bomLines: Ins<typeof s.bomLine>[]
  sizes: Ins<typeof s.size>[]
  products: Ins<typeof s.product>[]
  variants: Ins<typeof s.productVariant>[]
  sweetness: Ins<typeof s.sweetnessLevel>[]
  channels: Ins<typeof s.channel>[]
  prices: Ins<typeof s.price>[]
  recipes: Ins<typeof s.recipe>[]
  recipeLines: Ins<typeof s.recipeLine>[]
  equipment: Ins<typeof s.equipment>[]
}

export type SeedOpts = { newId: () => string; now: string; effectiveFrom: string }

function must<K, V>(m: Map<K, V>, k: K, what: string): V {
  const v = m.get(k)
  if (v === undefined) throw new Error(`seed: unknown ${what} ${String(k)}`)
  return v
}

export function seedToRows(seed: Seed, o: SeedOpts): SeedRows {
  const catId = new Map(seed.categories.map((c) => [c.code, o.newId()]))
  const itemId = new Map(seed.items.map((i) => [i.code, o.newId()]))
  const sizeId = new Map(seed.sizes.map((x) => [x.code, o.newId()]))
  const productId = new Map(seed.products.map((p) => [p.code, o.newId()]))
  const variantId = new Map(seed.variants.map((v) => [`${v.productCode}|${v.sizeCode}`, o.newId()]))
  const sweetId = new Map(seed.sweetness.map((x) => [x.code, o.newId()]))
  const channelId = new Map(seed.channels.map((x) => [x.code, o.newId()]))

  const rows: SeedRows = {
    categories: seed.categories.map((c) => ({ id: must(catId, c.code, 'category'), code: c.code, name: c.name, sort: c.sort, isActive: true })),
    items: seed.items.map((i) => ({
      id: must(itemId, i.code, 'item'), code: i.code, name: i.name, kind: i.kind, category: i.category, useUnit: i.useUnit,
      isTracked: i.isTracked, reorderPointMilli: i.reorderPointMilli, standardCostUsat: i.standardCostUsat, shelfLifeHours: i.shelfLifeHours, isActive: true, note: i.note,
    })),
    purchaseUnits: seed.purchaseUnits.map((u) => ({ id: o.newId(), itemId: must(itemId, u.itemCode, 'item'), name: u.name, qtyPerUnitMilli: u.qtyPerUnitMilli, isDefault: u.isDefault, barcode: null })),
    boms: [],
    bomLines: [],
    sizes: seed.sizes.map((x) => ({ id: must(sizeId, x.code, 'size'), code: x.code, name: x.name, sort: x.sort, packagingItemId: must(itemId, x.packagingItemCode, 'item') })),
    products: seed.products.map((p) => ({ id: must(productId, p.code, 'product'), code: p.code, nameTh: p.nameTh, nameEn: p.nameEn, categoryId: must(catId, p.categoryCode, 'category'), sort: p.sort, isActive: true, prepGroup: p.prepGroup, soldOutUntil: null })),
    variants: seed.variants.map((v) => ({ id: must(variantId, `${v.productCode}|${v.sizeCode}`, 'variant'), productId: must(productId, v.productCode, 'product'), sizeId: must(sizeId, v.sizeCode, 'size'), sku: v.sku, isActive: true })),
    sweetness: seed.sweetness.map((x) => ({ id: must(sweetId, x.code, 'sweetness'), code: x.code, name: x.name, sort: x.sort, isDefault: x.isDefault })),
    channels: seed.channels.map((x) => ({ id: must(channelId, x.code, 'channel'), code: x.code, name: x.name, commissionBp: x.commissionBp, isActive: true })),
    prices: seed.prices.map((p) => ({ id: o.newId(), variantId: must(variantId, `${p.productCode}|${p.sizeCode}`, 'variant'), channelId: must(channelId, p.channelCode, 'channel'), priceSatang: p.priceSatang, effectiveFrom: o.effectiveFrom, createdBy: null, createdAt: o.now })),
    recipes: [],
    recipeLines: [],
    equipment: seed.equipment.map((e) => ({ id: o.newId(), code: e.code, name: e.name, purchasedAt: e.purchasedAt, priceSatang: e.priceSatang, qty: e.qty, supplier: e.supplier, lifeYears: e.lifeYears === null ? null : Math.round(e.lifeYears), condition: e.condition, owner: e.owner, note: e.note })),
  }
  for (const b of seed.boms) {
    const bomId = o.newId()
    rows.boms.push({ id: bomId, itemId: must(itemId, b.itemCode, 'item'), version: 1, yieldMilli: b.yieldMilli, isCurrent: true, instructions: null })
    for (const l of b.lines) rows.bomLines.push({ id: o.newId(), bomId, componentItemId: must(itemId, l.itemCode, 'item'), qtyMilli: l.qtyMilli })
  }
  for (const r of seed.recipes) {
    const recipeId = o.newId()
    rows.recipes.push({ id: recipeId, variantId: must(variantId, `${r.productCode}|${r.sizeCode}`, 'variant'), sweetnessId: must(sweetId, r.sweetnessCode, 'sweetness'), version: 1, effectiveFrom: o.effectiveFrom, isCurrent: true, createdBy: null, note: null })
    for (const l of r.lines) rows.recipeLines.push({ id: o.newId(), recipeId, itemId: must(itemId, l.itemCode, 'item'), qtyMilli: l.qtyMilli })
  }
  return rows
}
```

import type { BetterSQLite3Database } from 'drizzle-orm/better-sqlite3'
import { eq } from 'drizzle-orm'
import { buildCatalog, type Catalog } from '@dayo/domain'
import * as s from '../sqlite/index.js'
import type { SeedRows } from './rows.js'

export function applySeedSqlite(db: BetterSQLite3Database, rows: SeedRows): void {
  db.transaction((tx) => {
    tx.insert(s.category).values(rows.categories).run()
    tx.insert(s.item).values(rows.items).run()
    tx.insert(s.purchaseUnit).values(rows.purchaseUnits).run()
    tx.insert(s.bom).values(rows.boms).run()
    tx.insert(s.bomLine).values(rows.bomLines).run()
    tx.insert(s.size).values(rows.sizes).run()
    tx.insert(s.product).values(rows.products).run()
    tx.insert(s.productVariant).values(rows.variants).run()
    tx.insert(s.sweetnessLevel).values(rows.sweetness).run()
    tx.insert(s.channel).values(rows.channels).run()
    tx.insert(s.price).values(rows.prices).run()
    tx.insert(s.recipe).values(rows.recipes).run()
    for (let i = 0; i < rows.recipeLines.length; i += 500) tx.insert(s.recipeLine).values(rows.recipeLines.slice(i, i + 500)).run()
    if (rows.equipment.length) tx.insert(s.equipment).values(rows.equipment).run()
  })
}

/** Domain catalog from the DB (current BOM versions only). */
export function loadCatalogSqlite(db: BetterSQLite3Database): Catalog {
  const items = db.select().from(s.item).all().map((i) => ({ id: i.id, kind: i.kind as 'raw' | 'prepared' | 'packaging_set', isTracked: i.isTracked, standardCostUsat: i.standardCostUsat }))
  const boms = db.select().from(s.bom).where(eq(s.bom.isCurrent, true)).all()
  const lines = db.select().from(s.bomLine).all()
  return buildCatalog(items, boms.map((b) => ({ itemId: b.itemId, yieldMilli: b.yieldMilli, lines: lines.filter((l) => l.bomId === b.id).map((l) => ({ itemId: l.componentItemId, qtyMilli: l.qtyMilli })) })))
}

import type { PgliteDatabase } from 'drizzle-orm/pglite'
import { eq } from 'drizzle-orm'
import { buildCatalog, type Catalog } from '@dayo/domain'
import * as p from '../pg/index.js'
import type { SeedRows } from './rows.js'

async function chunked<T>(rowsArr: T[], insert: (chunk: T[]) => Promise<unknown>): Promise<void> {
  for (let i = 0; i < rowsArr.length; i += 500) await insert(rowsArr.slice(i, i + 500))
}

export async function applySeedPg(db: PgliteDatabase, rows: SeedRows): Promise<void> {
  await db.transaction(async (tx) => {
    await tx.insert(p.category).values(rows.categories)
    await tx.insert(p.item).values(rows.items)
    await tx.insert(p.purchaseUnit).values(rows.purchaseUnits)
    await tx.insert(p.bom).values(rows.boms)
    await tx.insert(p.bomLine).values(rows.bomLines)
    await tx.insert(p.size).values(rows.sizes)
    await tx.insert(p.product).values(rows.products)
    await tx.insert(p.productVariant).values(rows.variants)
    await tx.insert(p.sweetnessLevel).values(rows.sweetness)
    await tx.insert(p.channel).values(rows.channels)
    await tx.insert(p.price).values(rows.prices)
    await tx.insert(p.recipe).values(rows.recipes)
    await chunked(rows.recipeLines, (c) => tx.insert(p.recipeLine).values(c))
    if (rows.equipment.length) await tx.insert(p.equipment).values(rows.equipment)
  })
}

export async function loadCatalogPg(db: PgliteDatabase): Promise<Catalog> {
  const items = (await db.select().from(p.item)).map((i) => ({ id: i.id, kind: i.kind as 'raw' | 'prepared' | 'packaging_set', isTracked: i.isTracked, standardCostUsat: i.standardCostUsat }))
  const boms = await db.select().from(p.bom).where(eq(p.bom.isCurrent, true))
  const lines = await db.select().from(p.bomLine)
  return buildCatalog(items, boms.map((b) => ({ itemId: b.itemId, yieldMilli: b.yieldMilli, lines: lines.filter((l) => l.bomId === b.id).map((l) => ({ itemId: l.componentItemId, qtyMilli: l.qtyMilli })) })))
}
