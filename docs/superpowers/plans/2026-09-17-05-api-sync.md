# แผน 5: api-sync — เซิร์ฟเวอร์ NestJS · ล็อกอินเครื่อง · push/pull · deploy Render/Neon · ping · สำรองข้อมูล — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** ทำให้ข้อมูลที่แท็บเล็ตบันทึกไว้ในเครื่อง (แผน 3 · 3b · 4) ขึ้นไปอยู่บนเซิร์ฟเวอร์จริงและถูกสำรองทุกคืน: `apps/api` (NestJS บน Render free สิงคโปร์ + Neon) รับ `POST /sync/push` แบบ idempotent พร้อม **ตรวจซ้ำทุกยอดด้วย `@dayo/domain` ชุดเดียวกัน** และตรวจโซ่แฮช · ส่งข้อมูลอ้างอิงลงเครื่องด้วย `GET /sync/pull?table=…&since=server_seq` · `/health` และ `/sync/inbox-flag` ไม่แตะ DB (ให้ Neon หลับได้) · ฝั่งแท็บเล็ตมีตัวส่ง/ดึงใน Worker ตามจังหวะ spec §6.2 · หน้า "สถานะระบบ" แสดง sync ล่าสุด · รายการที่เซิร์ฟเวอร์ปฏิเสธ (dead letter) พร้อมปุ่มส่งใหม่ · ลงทะเบียนเครื่องและกู้คืนเครื่องใหม่ · deploy + ping ทุก 10 นาที + `pg_dump` → R2 ทุกคืน

**Architecture:** เซิร์ฟเวอร์ใช้ `packages/db-schema` dialect `pg` และ `packages/domain` ตัวเดียวกับเครื่อง — ไม่มีสูตรคำนวณใหม่ฝั่งเซิร์ฟเวอร์เลย · payload ทุกชิ้นที่ข้าม network เป็น Zod schema ใน `packages/contracts` (spec §2.1) และมีเทสต์ parity เทียบกับคอลัมน์จริงของตาราง pg เพื่อไม่ให้ schema เลื่อนกันเงียบ ๆ · **ผู้เขียนคนเดียว** (T5-3): Render free รันอินสแตนซ์เดียว และทุกธุรกรรมเขียนวิ่งผ่าน mutex ในโปรเซส เหมือนคิว `serial()` ของ Worker ฝั่งเครื่อง → `server_seq` ที่ pull เห็นไม่มีวันมีช่องโหว่จากธุรกรรมที่ยังไม่ commit (ปิดประเด็นที่ spec §6.1 เขียนว่า "แผน 4 เลือก" — ส่งต่อมาถึงแผนนี้ตาม T4-14) · push แยกผลลัพธ์เป็น 4 สถานะต่อแถว (`accepted` · `duplicate` · `deferred` · `rejected`) → แถวที่ยังรอของที่มาก่อนหน้าไม่ตาย แต่แถวที่ผิดจริงไปเป็น dead letter ทันทีโดยไม่บล็อกแถวอื่น (spec §6.1) · ฝั่งเครื่องเพิ่มโมดูล `apps/pos/src/sync/` ที่ทำงานใน Worker เดียวกับ `PosApi` (ใช้คิว `serial()` เดิม = ไม่มีผู้เขียนตัวที่สองในเครื่อง — ปิดความเสี่ยง m-6 ของแผน 3b) และหน้าจอเรียกผ่านเมธอดใหม่ของ `PosApi` เท่านั้น

**Tech Stack:** ของเดิมทั้งหมดไม่เปลี่ยนเวอร์ชัน (Node 22 ≥ 22.16 · pnpm 10 · TypeScript ~5.9 · React 19.3.0 · Vite 8.3.0 · drizzle-orm 0.45.2 · Vitest 5.0.1 · Playwright 1.63.0 · zod ^4.6.0) · **เพิ่มเฉพาะใน `apps/api` (แพ็กเกจใหม่):** `@nestjs/common` `@nestjs/core` `@nestjs/platform-express` 11.x · `reflect-metadata` 0.2.x · `rxjs` 7.8.x · `pg` 8.x (driver ของ Neon ผ่าน TCP) · dev: `supertest` 7.x + `@types/supertest` · `@electric-sql/pglite` ^0.5.0 (มีอยู่แล้วใน db-schema — ใช้เป็นฐานข้อมูลของเทสต์ ไม่ต้องใช้ Docker) · `tsx` 4.23.13 · **ห้ามเพิ่ม dependency อื่นโดยไม่ถาม** (Q5-7 เพดานค่าใช้จ่าย = ฿0 · ยิ่ง dependency น้อย ยิ่ง cold start เร็ว)

**Spec:** [../specs/2026-09-17-pos-design.md](../specs/2026-09-17-pos-design.md) §2 (สถาปัตยกรรม · โครง monorepo · อุปกรณ์), §3 (R/T/L), §3.6 (`outbox` · `sync_state` · `idempotency_record` · `server_cursor`/`server_seq` · `invariant_run`), §4.9 (แฮชโซ่), §6.1 (push/pull/dead-letter/เซิร์ฟเวอร์ชนะ), §6.2 (จังหวะ sync), §6.3 (นาฬิกา), §6.4 (ติดตั้ง/กู้คืนเครื่องใหม่), §7 (Auth · Sync · Health — Admin/Reports/Jobs/LINE **ไม่อยู่ในแผนนี้** ดูหัวข้อ "ขอบเขต"), §8 (Unit: sync push/pull idempotency ส่งซ้ำ 3 ครั้งได้ผลเดียว), §10 (โฮสติ้ง · CI · backup · migration · monitoring · ทางหนี), §11 ขั้น 5, §12 (ความเสี่ยง: แท็บเล็ตพังก่อน sync · Neon เกิน 100 CU-ชม. · Render หลับ) · การตัดสินใจ: [../../design/00-บันทึกการตัดสินใจ.md](../../design/00-บันทึกการตัดสินใจ.md) โดยเฉพาะ D6, D13, D14, **D15 (โฮสติ้งฟรี ไม่ผูกบัตร)**, D21, D33, D42, **D43 (โมเดล agent)**, **D45 (commit)**, D47 ข้อ 8–9, **D49 (ห้าม push)**, D51, D57 · งานส่งต่อที่แผนนี้ต้องรับ: [บันทึกแผน 3 §3, §5](2026-09-17-03-บันทึกการทำแผน3.md) · [บันทึกแผน 3b §5, §6](2026-09-17-03b-บันทึกการทำแผน3b.md) · [บันทึกแผน 4 §5, §6](2026-09-17-04-บันทึกการทำแผน4.md) (ตารางเช็กครบอยู่ท้ายแผน) · **คำถามก่อนทำ: [2026-09-17-05-คำถามก่อนทำ.md](2026-09-17-05-คำถามก่อนทำ.md) — 17 ข้อ (Q5-1 … Q5-17) รอเจ้าของตอบ** · แผนนี้เขียนตาม **ค่าแนะนำ** ทุกข้อ · การตัดสินใจเชิงเทคนิค T5-1 … T5-18 อยู่ท้ายไฟล์คำถาม

```ts
import { computeEventHash, GENESIS_HASH, type EventCore } from './hash.js'

/**
 * Server-side re-verification (spec §7: "ใช้ packages/domain ชุดเดียวกัน ตรวจซ้ำทุกยอดที่รับมา").
 * Pure functions: the API calls them with rows it has already stored or is about to store.
 */

export type VerifyResult = { ok: true } | { ok: false; reason: string }

export type OrderTotalsInput = {
  subtotalSatang: number
  discountSatang: number
  totalSatang: number
  status: string
  lineTotals: readonly number[]
  paymentAmounts: readonly number[]
}

/** spec §7 invariant 1. `paid` covers the statuses where money has been taken (paid · ready · picked_up · voided). */
const PAID_STATUSES = new Set(['paid', 'ready', 'picked_up', 'voided'])

export function verifyOrderTotals(o: OrderTotalsInput): VerifyResult {
  const sumLines = o.lineTotals.reduce((a, b) => a + b, 0)
  if (sumLines !== o.subtotalSatang) return { ok: false, reason: `sum(line_total) ${sumLines} != subtotal ${o.subtotalSatang}` }
  if (o.subtotalSatang - o.discountSatang !== o.totalSatang) {
    return { ok: false, reason: `subtotal ${o.subtotalSatang} - discount ${o.discountSatang} != total ${o.totalSatang}` }
  }
  if (o.discountSatang >= o.subtotalSatang && o.subtotalSatang > 0) {
    return { ok: false, reason: `discount ${o.discountSatang} must be less than subtotal ${o.subtotalSatang}` } // D50 Q3-20
  }
  if (PAID_STATUSES.has(o.status)) {
    const sumPaid = o.paymentAmounts.reduce((a, b) => a + b, 0)
    if (sumPaid !== o.totalSatang) return { ok: false, reason: `sum(payment) ${sumPaid} != total ${o.totalSatang} on a paid bill` }
  }
  return { ok: true }
}

export type ChainTip = { chainSeq: number; hash: string } | null
export type ChainLinkVerdict = { ok: true } | { ok: false; kind: 'gap' | 'broken'; reason: string }

/**
 * One link of a device's event chain (spec §4.9, D38). `gap` means an earlier event has not arrived yet — the caller
 * defers the row and retries later. `broken` means the content or the chaining is wrong — the caller kills the row.
 */
export function verifyChainLink(tip: ChainTip, e: EventCore & { prevHash: string; hash: string }): ChainLinkVerdict {
  const expectedSeq = tip === null ? 1 : tip.chainSeq + 1
  if (e.chainSeq < expectedSeq) return { ok: false, kind: 'broken', reason: `chain_seq ${e.chainSeq} is behind the stored tip ${tip?.chainSeq ?? 0}` }
  if (e.chainSeq > expectedSeq) return { ok: false, kind: 'gap', reason: `expected chain_seq ${expectedSeq}, got ${e.chainSeq}` }
  if (tip === null && e.prevHash !== GENESIS_HASH) return { ok: false, kind: 'broken', reason: 'a chain must start at the genesis hash' }
  if (tip !== null && e.prevHash !== tip.hash) return { ok: false, kind: 'broken', reason: 'prev_hash does not match the stored chain tip' }
  if (computeEventHash(e.prevHash, e) !== e.hash) return { ok: false, kind: 'broken', reason: 'hash does not match the event content' }
  return { ok: true }
}

/** Free text a person typed: the limit is counted in code points (plan 4 final review m-4). */
export const TEXT_MAX_CODE_POINTS = 200

export function textWithinLimit(s: string, max: number = TEXT_MAX_CODE_POINTS): boolean {
  return [...s].length <= max
}
```

```ts
import type { NodePgDatabase } from 'drizzle-orm/node-postgres'
import type { pg } from '@dayo/db-schema'

/** The one database type the whole API uses. Tests pass a PGlite drizzle instance cast to this type: both drivers
 *  implement the same query builder and the API never calls anything driver-specific. */
export type Db = NodePgDatabase<typeof pg>

export const DB = Symbol('DB')
export const WRITE_QUEUE = Symbol('WRITE_QUEUE')

/** Single-writer guarantee (T5-3): every write transaction runs one at a time inside this process, so a pull never
 *  sees a server_seq gap left by an in-flight transaction (spec §6.1). Same idea as the device's serial() queue. */
export type WriteQueue = <T>(fn: () => Promise<T>) => Promise<T>

export function createWriteQueue(): WriteQueue {
  let tail: Promise<unknown> = Promise.resolve()
  return <T>(fn: () => Promise<T>): Promise<T> => {
    const run = tail.then(fn, fn)
    tail = run.then(() => undefined, () => undefined)
    return run
  }
}
```

```ts
import { BadRequestException } from '@nestjs/common'
import type { ZodType } from 'zod'

/** Parses a body/query with zod and turns a failure into the shape the device understands. */
export function parseOrThrow<T>(schema: ZodType<T>, value: unknown): T {
  const parsed = schema.safeParse(value)
  if (!parsed.success) {
    const detail = parsed.error.issues.slice(0, 3).map((i) => `${i.path.join('.') || '(root)'}: ${i.message}`).join(' · ')
    throw new BadRequestException({ error: 'SCHEMA', detail })
  }
  return parsed.data
}
```

```ts
import { ConflictException, Inject, Injectable, UnauthorizedException } from '@nestjs/common'
import { and, eq, isNull, ne } from 'drizzle-orm'
import { pg as s } from '@dayo/db-schema'
import type { RegisterDeviceRequest, RegisterDeviceResponse } from '@dayo/contracts'
import { CONFIG, type AppConfig } from '../config.js'
import { DB, WRITE_QUEUE, type Db, type WriteQueue } from '../db/types.js'
import { hashToken, newDeviceToken } from './tokens.js'

export type DeviceIdentity = { deviceId: string; deviceName: string }

@Injectable()
export class AuthService {
  constructor(
    @Inject(DB) private readonly db: Db,
    @Inject(WRITE_QUEUE) private readonly write: WriteQueue,
    @Inject(CONFIG) private readonly config: AppConfig,
  ) {}

  /**
   * Q5-9 · Q5-10 · spec §6.4: the tablet's setup screen already created the device, the owners (pin_hash only) and
   * the PromptPay setting locally; this call copies those rows up once and hands back a long-lived token. From then
   * on the server owns them (type R) and the device pulls them back down.
   */
  async register(req: RegisterDeviceRequest): Promise<RegisterDeviceResponse> {
    if (req.registrationCode !== this.config.registrationCode) throw new UnauthorizedException({ error: 'BAD_CODE' })
    const token = newDeviceToken()
    const tokenHash = hashToken(token)
    const at = new Date().toISOString()

    await this.write(async () => {
      const clash = await this.db.select({ id: s.device.id }).from(s.device)
        .where(and(eq(s.device.receiptPrefix, req.receiptPrefix), ne(s.device.id, req.deviceId))).limit(1)
      if (clash.length > 0) throw new ConflictException({ error: 'PREFIX_TAKEN', detail: req.receiptPrefix })

      await this.db.transaction(async (tx) => {
        await tx.insert(s.device)
          .values({ id: req.deviceId, name: req.deviceName, receiptPrefix: req.receiptPrefix, isSellingDevice: true, registeredAt: at, updatedAt: at, version: 1 })
          .onConflictDoUpdate({ target: s.device.id, set: { name: req.deviceName, updatedAt: at } })
        for (const u of req.users) {
          await tx.insert(s.user)
            .values({ id: u.id, displayName: u.displayName, role: u.role, pinHash: u.pinHash, isActive: u.isActive, createdAt: u.createdAt, updatedAt: at, version: 1 })
            .onConflictDoNothing({ target: s.user.id })
        }
        for (const st of req.settings) {
          await tx.insert(s.setting)
            .values({ key: st.key, valueJson: st.valueJson, effectiveFrom: st.effectiveFrom, updatedAt: at, version: 1 })
            .onConflictDoNothing({ target: [s.setting.key, s.setting.effectiveFrom] })
        }
        await tx.insert(s.deviceToken).values({ id: `tok-${tokenHash.slice(0, 24)}`, deviceId: req.deviceId, tokenHash, createdAt: at, lastUsedAt: null, revokedAt: null })
        await tx.insert(s.auditLog).values({
          id: `aud-${tokenHash.slice(0, 24)}`, entity: 'device', entityId: req.deviceId, action: 'register',
          beforeJson: null, afterJson: { name: req.deviceName, receiptPrefix: req.receiptPrefix, users: req.users.length, settings: req.settings.length },
          actorUserId: req.users[0]?.id ?? null, at,
        })
      })
    })

    return { deviceId: req.deviceId, token, serverTime: at }
  }

  /** Used by POST /auth/device and by DeviceGuard on every sync call. */
  async authenticate(deviceId: string, token: string): Promise<DeviceIdentity> {
    const rows = await this.db
      .select({ deviceId: s.deviceToken.deviceId, name: s.device.name })
      .from(s.deviceToken)
      .innerJoin(s.device, eq(s.device.id, s.deviceToken.deviceId))
      .where(and(eq(s.deviceToken.tokenHash, hashToken(token)), eq(s.deviceToken.deviceId, deviceId), isNull(s.deviceToken.revokedAt)))
      .limit(1)
    const row = rows[0]
    if (!row) throw new UnauthorizedException({ error: 'BAD_TOKEN' })
    return { deviceId: row.deviceId, deviceName: row.name }
  }
}
```

```ts
import { CanActivate, createParamDecorator, ExecutionContext, Injectable, UnauthorizedException } from '@nestjs/common'
import type { Request } from 'express'
import { AuthService, type DeviceIdentity } from './auth.service.js'

type AuthedRequest = Request & { device?: DeviceIdentity }

@Injectable()
export class DeviceGuard implements CanActivate {
  constructor(private readonly auth: AuthService) {}

  async canActivate(ctx: ExecutionContext): Promise<boolean> {
    const req = ctx.switchToHttp().getRequest<AuthedRequest>()
    const header = req.headers.authorization ?? ''
    const token = header.startsWith('Bearer ') ? header.slice('Bearer '.length) : ''
    const deviceId = String(req.headers['x-device-id'] ?? '')
    if (token === '' || deviceId === '') throw new UnauthorizedException({ error: 'BAD_TOKEN' })
    req.device = await this.auth.authenticate(deviceId, token)
    return true
  }
}

export const CurrentDevice = createParamDecorator((_data: unknown, ctx: ExecutionContext): DeviceIdentity => {
  const req = ctx.switchToHttp().getRequest<AuthedRequest>()
  if (!req.device) throw new UnauthorizedException({ error: 'BAD_TOKEN' })
  return req.device
})
```

```ts
import { Inject, Injectable, Logger } from '@nestjs/common'
import { and, asc, eq, inArray, sql } from 'drizzle-orm'
import { pg as s } from '@dayo/db-schema'
import { SYNC_ROW_SCHEMAS, UPSERTABLE_TABLES, type PushRequest, type PushResponse, type RowResult, type SyncTable } from '@dayo/contracts'
import { canonicalJson, rebuildCostState, sha256Hex, verifyChainLink, verifyOrderTotals } from '@dayo/domain'
import { DB, WRITE_QUEUE, type Db, type WriteQueue } from '../db/types.js'
import { DEVICE_FIELD_OF, PARENT_OF, PG_TABLE_OF } from './tables.js'

type Verdict = Omit<RowResult, 'idempotencyKey'>

const PG_FK_VIOLATION = '23503'

@Injectable()
export class PushService {
  private readonly log = new Logger(PushService.name)

  constructor(@Inject(DB) private readonly db: Db, @Inject(WRITE_QUEUE) private readonly write: WriteQueue) {}

  /**
   * spec §6.1: insert-if-absent per row, re-verify with the domain, answer per row. One bad row never blocks the
   * rows around it. Every write runs inside the single-writer queue (T5-3), so `server_seq` values become visible
   * in order and a pull can never skip a row that is still being written.
   */
  async ingest(deviceId: string, req: PushRequest): Promise<PushResponse> {
    const results: RowResult[] = []
    const touchedItems = new Set<string>()

    await this.write(async () => {
      for (const pushed of req.rows) {
        const verdict = await this.ingestRow(deviceId, pushed.idempotencyKey, pushed.table, pushed.row, touchedItems)
        results.push({ idempotencyKey: pushed.idempotencyKey, ...verdict })
      }
      if (touchedItems.size > 0) await this.rebuildCostStates([...touchedItems])
    })

    return { serverTime: new Date().toISOString(), results }
  }

  private async ingestRow(deviceId: string, key: string, table: SyncTable, raw: unknown, touchedItems: Set<string>): Promise<Verdict> {
    const parsed = SYNC_ROW_SCHEMAS[table].safeParse(raw)
    if (!parsed.success) {
      const issue = parsed.error.issues[0]
      return { status: 'rejected', reason: 'SCHEMA', detail: `${issue?.path.join('.') ?? '(root)'}: ${issue?.message ?? 'invalid'}` }
    }
    const row = parsed.data as Record<string, unknown> & { id: string }
    const contentHash = sha256Hex(canonicalJson(row))
    const rowKey = `${table}:${row.id}`

    const deviceField = DEVICE_FIELD_OF[table]
    if (deviceField !== undefined) {
      const owner = row[deviceField]
      if (owner !== null && owner !== deviceId) return { status: 'rejected', reason: 'NOT_MY_DEVICE', detail: String(owner) }
    }

    const seen = await this.db.select().from(s.ingestedRow).where(eq(s.ingestedRow.key, rowKey)).limit(1)
    const before = seen[0]
    if (before !== undefined && before.contentHash === contentHash) return { status: 'duplicate' }
    if (before !== undefined && !UPSERTABLE_TABLES.has(table)) {
      return { status: 'rejected', reason: 'CONFLICT', detail: `${rowKey} was already stored with different content` }
    }

    const parent = PARENT_OF[table]
    if (parent !== undefined) {
      const parentId = row[parent.field]
      if (typeof parentId === 'string') {
        const exists = await this.db.select({ id: s.ingestedRow.rowId }).from(s.ingestedRow).where(eq(s.ingestedRow.key, `${parent.table}:${parentId}`)).limit(1)
        if (exists.length === 0) return { status: 'deferred', reason: 'PARENT_PENDING', detail: `${parent.table}:${parentId}` }
      }
    }

    if (table === 'order_event') {
      const verdict = await this.checkChain(row)
      if (verdict !== null) return verdict
    }

    try {
      await this.db.transaction(async (tx) => {
        const target = PG_TABLE_OF[table]
        const insert = tx.insert(target).values(row as never)
        if (UPSERTABLE_TABLES.has(table)) {
          await insert.onConflictDoUpdate({ target: (target as unknown as { id: never }).id, set: row as never })
        } else {
          await insert.onConflictDoNothing()
        }
        await tx.insert(s.ingestedRow)
          .values({ key: rowKey, tableName: table, rowId: row.id, contentHash, firstSeenAt: new Date().toISOString() })
          .onConflictDoUpdate({ target: s.ingestedRow.key, set: { contentHash } })
        await tx.insert(s.idempotencyRecord)
          .values({ key, firstSeenAt: new Date().toISOString(), resultHash: contentHash })
          .onConflictDoNothing()
      })
    } catch (e) {
      const code = (e as { cause?: { code?: string }; code?: string }).code ?? (e as { cause?: { code?: string } }).cause?.code
      if (code === PG_FK_VIOLATION) return { status: 'rejected', reason: 'FK_MISSING', detail: String((e as Error).message).slice(0, 200) }
      this.log.error(`ingest ${rowKey} failed`, e as Error)
      throw e
    }

    if (table === 'stock_movement') touchedItems.add(String(row['itemId']))

    // spec §7 invariant 1 — re-check the whole bill once its parts are in (a bill may arrive across pushes).
    if (table === 'payment' || table === 'order' || table === 'order_line' || table === 'discount') {
      const orderId = String(table === 'order' ? row['id'] : row['orderId'])
      const verdict = await this.checkOrderTotals(orderId)
      if (verdict !== null) return verdict
    }
    return { status: 'accepted' }
  }

  /** A gap means an earlier event has not arrived (retry later); a hash mismatch means the row is wrong for good. */
  private async checkChain(row: Record<string, unknown>): Promise<Verdict | null> {
    const chainId = String(row['chainId'])
    const tipRows = await this.db.select({ chainSeq: s.orderEvent.chainSeq, hash: s.orderEvent.hash })
      .from(s.orderEvent).where(eq(s.orderEvent.chainId, chainId))
      .orderBy(sql`${s.orderEvent.chainSeq} desc`).limit(1)
    const tip = tipRows[0] ?? null
    const verdict = verifyChainLink(tip, row as never)
    if (verdict.ok) return null
    return verdict.kind === 'gap'
      ? { status: 'deferred', reason: 'CHAIN_GAP', detail: verdict.reason }
      : { status: 'rejected', reason: 'CHAIN_BROKEN', detail: verdict.reason }
  }

  /** Runs on the rows already stored, so it also catches a bill assembled from two pushes. */
  private async checkOrderTotals(orderId: string): Promise<Verdict | null> {
    const orders = await this.db.select().from(s.order).where(eq(s.order.id, orderId)).limit(1)
    const order = orders[0]
    if (!order) return null
    const lines = await this.db.select({ total: s.orderLine.lineTotalSatang }).from(s.orderLine).where(eq(s.orderLine.orderId, orderId))
    if (lines.length === 0) return null                      // the lines have not arrived yet — nothing to check
    const payments = await this.db.select({ amount: s.payment.amountSatang }).from(s.payment).where(eq(s.payment.orderId, orderId))
    const verdict = verifyOrderTotals({
      subtotalSatang: order.subtotalSatang, discountSatang: order.discountSatang, totalSatang: order.totalSatang,
      status: order.status, lineTotals: lines.map((l) => l.total), paymentAmounts: payments.map((p) => p.amount),
    })
    if (verdict.ok) return null
    if (payments.length === 0) return null                   // payments still on their way — do not judge yet
    return { status: 'rejected', reason: 'TOTALS', detail: verdict.reason }
  }

  /**
   * The device never pushes `item_cost_state` (plan 3 §5): the server rebuilds it from the movements it has, with
   * the same domain function the tablet uses, so the moving average and D57's per-document grouping stay identical.
   */
  private async rebuildCostStates(itemIds: string[]): Promise<void> {
    const items = await this.db.select({ id: s.item.id, standard: s.item.standardCostUsat }).from(s.item).where(inArray(s.item.id, itemIds))
    const at = new Date().toISOString()
    for (const item of items) {
      const movements = await this.db
        .select({ qtyMilli: s.stockMovement.qtyMilli, unitCostUsat: s.stockMovement.unitCostUsat, refType: s.stockMovement.refType, refId: s.stockMovement.refId })
        .from(s.stockMovement).where(eq(s.stockMovement.itemId, item.id))
        .orderBy(asc(s.stockMovement.createdAt), asc(s.stockMovement.id))
      const state = rebuildCostState(movements, item.standard)
      await this.db.insert(s.itemCostState)
        .values({ itemId: item.id, onHandMilli: state.onHandMilli, avgCostUsat: state.avgCostUsat, updatedAt: at })
        .onConflictDoUpdate({ target: s.itemCostState.itemId, set: { onHandMilli: state.onHandMilli, avgCostUsat: state.avgCostUsat, updatedAt: at } })
    }
  }
}
```

```ts
import { Inject, Injectable } from '@nestjs/common'
import { asc, getTableColumns, gt, type PgTable } from 'drizzle-orm/pg-core'
import { pg as s } from '@dayo/db-schema'
import type { PullQuery, PullResponse, PullTable } from '@dayo/contracts'
import { DB, type Db } from '../db/types.js'

export const PULL_TABLE_OF: Record<PullTable, PgTable> = {
  user: s.user, device: s.device, setting: s.setting, category: s.category, item: s.item,
  purchase_unit: s.purchaseUnit, bom: s.bom, bom_line: s.bomLine, product: s.product, size: s.size,
  product_variant: s.productVariant, sweetness_level: s.sweetnessLevel, channel: s.channel, price: s.price,
  recipe: s.recipe, recipe_line: s.recipeLine, equipment: s.equipment, customer: s.customer,
  order: s.order, order_line: s.orderLine, order_event: s.orderEvent, payment: s.payment,
  item_cost_state: s.itemCostState,
}

/** Columns the device has no column for — they exist only on the server side of the schema. */
const SERVER_ONLY = new Set(['serverSeq', 'serverReceivedAt'])

@Injectable()
export class PullService {
  constructor(@Inject(DB) private readonly db: Db) {}

  /**
   * spec §6.1: `GET /sync/pull?table=…&since=server_seq`. A row edited on the server gets a fresh server_seq from
   * the `bump_server_seq` trigger (plan 2), so an edit is pulled again. Reading is safe without a snapshot guard
   * because this process is the only writer (T5-3) and every write runs inside WriteQueue: by the time a row is
   * visible, every smaller server_seq is committed too.
   */
  async pull(q: PullQuery): Promise<PullResponse> {
    const table = PULL_TABLE_OF[q.table]
    const columns = getTableColumns(table)
    const seq = columns['serverSeq']
    if (seq === undefined) throw new Error(`table ${q.table} has no server_seq`)   // unreachable: every pull table has one
    const selection = Object.fromEntries(Object.entries(columns).filter(([name]) => !SERVER_ONLY.has(name)))

    const rows = await this.db
      .select({ ...selection, __seq: seq })
      .from(table)
      .where(gt(seq, q.since))
      .orderBy(asc(seq))
      .limit(q.limit + 1)

    const hasMore = rows.length > q.limit
    const page = hasMore ? rows.slice(0, q.limit) : rows
    const last = page.at(-1) as { __seq?: number } | undefined
    const nextSince = last?.__seq ?? q.since
    return {
      serverTime: new Date().toISOString(),
      table: q.table,
      rows: page.map((r) => { const { __seq: _drop, ...rest } = r as Record<string, unknown> & { __seq: number }; return rest }),
      nextSince,
      hasMore,
    }
  }
}
```

```ts
export const TEST_SERVER = 'http://api.test'

export type FakeServer = {
  fetch: typeof globalThis.fetch
  /** Every push body the device sent, in order. */
  pushes: PushRequest[]
  /** Decide the answer for one row; default: accept everything. */
  verdict: (row: PushRequest['rows'][number], index: number) => Omit<RowResult, 'idempotencyKey'>
  /** Rows the fake server hands back on a pull, by table. */
  pullRows: Map<string, { rows: Record<string, unknown>[]; nextSince: number }>
  serverTime: string
  offline: boolean
  unauthorized: boolean
  calls: string[]
}

export function fakeServer(): FakeServer {
  const f: FakeServer = {
    fetch: (async () => new Response('', { status: 500 })) as typeof globalThis.fetch,
    pushes: [],
    verdict: () => ({ status: 'accepted' }),
    pullRows: new Map(),
    serverTime: '2026-09-22T03:00:05.000Z',
    offline: false,
    unauthorized: false,
    calls: [],
  }
  f.fetch = (async (input: RequestInfo | URL, init?: RequestInit) => {
    const url = new URL(String(input))
    f.calls.push(`${init?.method ?? 'GET'} ${url.pathname}${url.search}`)
    if (f.offline) throw new TypeError('Failed to fetch')
    if (f.unauthorized) return new Response(JSON.stringify({ error: 'BAD_TOKEN' }), { status: 401, headers: { 'content-type': 'application/json' } })
    if (url.pathname === '/sync/push') {
      const body = JSON.parse(String(init?.body)) as PushRequest
      f.pushes.push(body)
      const results = body.rows.map((r, i) => ({ idempotencyKey: r.idempotencyKey, ...f.verdict(r, i) }))
      return new Response(JSON.stringify({ serverTime: f.serverTime, results }), { status: 201, headers: { 'content-type': 'application/json' } })
    }
    if (url.pathname === '/sync/pull') {
      const table = url.searchParams.get('table') ?? ''
      const since = Number(url.searchParams.get('since') ?? '0')
      const prepared = f.pullRows.get(table) ?? { rows: [], nextSince: since }
      return new Response(JSON.stringify({ serverTime: f.serverTime, table, rows: prepared.rows, nextSince: prepared.nextSince, hasMore: false }), { status: 200, headers: { 'content-type': 'application/json' } })
    }
    if (url.pathname === '/sync/inbox-flag') {
      return new Response(JSON.stringify({ hasInbox: false, serverTime: f.serverTime }), { status: 200, headers: { 'content-type': 'application/json' } })
    }
    if (url.pathname === '/health') {
      return new Response(JSON.stringify({ ok: true, serverTime: f.serverTime, version: 'test' }), { status: 200, headers: { 'content-type': 'application/json' } })
    }
    return new Response('not found', { status: 404 })
  }) as typeof globalThis.fetch
  return f
}

/** Marks the device as registered so the transport has somewhere to call. */

  }
}
```

```ts
export class SyncHttpError extends Error {
  constructor(readonly status: number, message: string) { super(message); this.name = 'SyncHttpError' }
}

export type SyncTransport = {
  deviceId: string
  push: (body: PushRequest) => Promise<PushResponse>
  pull: (table: PullTable, since: number) => Promise<PullResponse>
  inboxFlag: () => Promise<InboxFlagResponse>
}

async function readValue(db: RemoteDb, key: string): Promise<string | null> {
  const row = await db.select().from(s.syncState).where(eq(s.syncState.key, key)).get()
  return row?.value ?? null
}

/** null when this tablet has not been registered with a server yet (Task 10). */
export async function createTransport(db: RemoteDb, deps: ApiDeps): Promise<SyncTransport | null> {
  const [baseUrl, token, deviceId] = await Promise.all([readValue(db, SYNC_KEYS.baseUrl), readValue(db, SYNC_KEYS.token), localDeviceId(db)])
  if (baseUrl === null || token === null || deviceId === null) return null
  const headers = { 'content-type': 'application/json', authorization: `Bearer ${token}`, 'x-device-id': deviceId }

  async function call(path: string, init?: RequestInit): Promise<unknown> {
    const res = await deps.fetch(`${baseUrl}${path}`, { ...init, headers })
    if (!res.ok) throw new SyncHttpError(res.status, `${res.status} ${await res.text().catch(() => '')}`.slice(0, 300))
    return res.json()
  }

  return {
    deviceId,
    push: async (body) => PushResponse.parse(await call('/sync/push', { method: 'POST', body: JSON.stringify(body) })),
    pull: async (table, since) => PullResponse.parse(await call(`/sync/pull?table=${table}&since=${since}`)),
    inboxFlag: async () => (await call('/sync/inbox-flag?clear=1')) as InboxFlagResponse,
  }
}
```

```ts
import { asc, eq } from 'drizzle-orm'
import type { RemoteDb } from '@dayo/db-schema/browser'
import * as s from '@dayo/db-schema/sqlite'
import { MAX_PUSH_ROWS, type PushRow, type SyncTable } from '@dayo/contracts'
import type { ApiDeps } from '../api/deps'
import type { SyncTransport } from './transport'

export type PushOutcome = { sent: number; dead: number; deferred: number }

/**
 * spec §6.1: FIFO, at most 200 rows per request, one idempotency key per row (the key the writer stored — plan 3 M7).
 * `accepted`/`duplicate` → sent · `rejected` → dead letter with the reason (never retried automatically, Q5-12) ·
 * `deferred` → stays pending and is retried on the next run.
 */
export async function pushOnce(db: RemoteDb, deps: ApiDeps, transport: SyncTransport): Promise<PushOutcome> {
  const outcome: PushOutcome = { sent: 0, dead: 0, deferred: 0 }
  for (;;) {
    const batch = await db.select().from(s.outbox).where(eq(s.outbox.status, 'pending')).orderBy(asc(s.outbox.createdAt), asc(s.outbox.id)).limit(MAX_PUSH_ROWS).all()
    if (batch.length === 0) return outcome

    const rows: PushRow[] = batch.map((r) => ({ idempotencyKey: r.idempotencyKey, table: r.tableName as SyncTable, row: r.rowJson as Record<string, unknown> }))
    const answer = await transport.push({ deviceId: transport.deviceId, rows })
    const verdictOf = new Map(answer.results.map((r) => [r.idempotencyKey, r]))
    const at = deps.now()

    for (const queued of batch) {
      const verdict = verdictOf.get(queued.idempotencyKey)
      if (verdict === undefined || verdict.status === 'deferred') {
        await db.update(s.outbox).set({ attempts: queued.attempts + 1, lastError: verdict === undefined ? 'no answer for this row' : `${verdict.reason ?? ''}: ${verdict.detail ?? ''}` }).where(eq(s.outbox.id, queued.id))
        outcome.deferred++
      } else if (verdict.status === 'rejected') {
        await db.update(s.outbox).set({ status: 'dead', deadAt: at, attempts: queued.attempts + 1, lastError: `${verdict.reason ?? 'rejected'}: ${verdict.detail ?? ''}`.slice(0, 500) }).where(eq(s.outbox.id, queued.id))
        outcome.dead++
      } else {
        await db.update(s.outbox).set({ status: 'sent', sentAt: at, attempts: queued.attempts + 1, lastError: null }).where(eq(s.outbox.id, queued.id))
        outcome.sent++
      }
    }
    // Everything left in this batch is deferred, so another round would send the same rows again: stop here.
    if (outcome.deferred > 0 && outcome.sent === 0 && outcome.dead === 0) return outcome
    if (batch.length < MAX_PUSH_ROWS) return outcome
  }
}
```

```ts
/** Transaction tables the device owns: a pulled row is inserted only if the device does not have it (spec §6.1 —
 *  the server wins on reference data, but it must never rewrite a bill this tablet is the source of). */
const INSERT_ONLY: ReadonlySet<PullTable> = new Set<PullTable>(['order', 'order_line', 'order_event', 'payment'])

export async function pullOnce(db: RemoteDb, transport: SyncTransport, tables: readonly PullTable[] = PULL_TABLES): Promise<PullOutcome> {
  let pulled = 0
  for (const table of tables) {
    const cursorKey = SYNC_KEYS.cursor(table)
    const cursorRow = await db.select().from(s.syncState).where(eq(s.syncState.key, cursorKey)).get()
    let since = Number(cursorRow?.value ?? '0')
    for (;;) {
      const page = await transport.pull(table, since)
      const target = SQLITE_TABLE_OF[table]
      await db.transaction(async (tx) => {
        for (const row of page.rows) {
          const insert = tx.insert(target.table).values(row as never)
          if (INSERT_ONLY.has(table)) await insert.onConflictDoNothing()
          else await insert.onConflictDoUpdate({ target: target.conflict, set: row as never })
        }
        const value = String(page.nextSince)
        await tx.insert(s.syncState).values({ key: cursorKey, value }).onConflictDoUpdate({ target: s.syncState.key, set: { value } })
      })
      pulled += page.rows.length
      since = page.nextSince
      if (!page.hasMore) break
    }
  }
  return { pulled }
}
```

```ts
import { eq } from 'drizzle-orm'
import type { RemoteDb } from '@dayo/db-schema/browser'
import * as s from '@dayo/db-schema/sqlite'
import { PULL_TABLES } from '@dayo/contracts'
import type { ApiDeps } from '../api/deps'
import type { SyncRunDto, SyncTickKind } from '../api/types'
import { CLOCK_SKEW_WARN_MS, SYNC_KEYS } from './keys'
import { pullOnce } from './pull'
import { pushOnce } from './push'
import { createTransport, SyncHttpError } from './transport'

async function putState(db: RemoteDb, key: string, value: string): Promise<void> {
  await db.insert(s.syncState).values({ key, value }).onConflictDoUpdate({ target: s.syncState.key, set: { value } })
}

const EMPTY: SyncRunDto = { ok: false, pushed: 0, dead: 0, deferred: 0, pulled: 0, clockSkewMs: 0, error: null }

/** Turns any transport failure into a code the screen can show in Thai (never an English message — constraints). */
function errorCode(e: unknown): SyncRunDto['error'] {
  if (e instanceof SyncHttpError) return e.status === 401 || e.status === 403 ? 'SYNC_UNAUTHORIZED' : 'SYNC_OFFLINE'
  return 'SYNC_OFFLINE'
}

export async function syncNow(db: RemoteDb, deps: ApiDeps, kind: SyncTickKind = 'both'): Promise<SyncRunDto> {
  const transport = await createTransport(db, deps)
  if (transport === null) return { ...EMPTY, error: 'NOT_REGISTERED' }

  const run: SyncRunDto = { ...EMPTY, ok: true, error: null }
  try {
    if (kind === 'flag') {
      const flag = await transport.inboxFlag()
      run.clockSkewMs = Date.parse(deps.now()) - Date.parse(flag.serverTime)
      if (!flag.hasInbox) return run
      const pulled = await pullOnce(db, transport, ['order', 'order_line', 'order_event', 'payment'])
      run.pulled = pulled.pulled
      return run
    }

    if (kind === 'push' || kind === 'both') {
      const pushed = await pushOnce(db, deps, transport)
      run.pushed = pushed.sent
      run.dead = pushed.dead
      run.deferred = pushed.deferred
      await putState(db, SYNC_KEYS.lastPushAt, deps.now())
    }

    if (kind === 'pull' || kind === 'both') {
      const pulled = await pullOnce(db, transport, PULL_TABLES)
      run.pulled = pulled.pulled
      await putState(db, SYNC_KEYS.lastPullAt, deps.now())
      const serverTime = (await transport.inboxFlag()).serverTime          // cheap, database-free clock reference
      run.clockSkewMs = Date.parse(deps.now()) - Date.parse(serverTime)
      await putState(db, SYNC_KEYS.lastServerTime, serverTime)
      await putState(db, SYNC_KEYS.skewMs, String(run.clockSkewMs))
    }
    return run
  } catch (e) {
    return { ...run, ok: false, error: errorCode(e) }
  }
}

export async function syncTick(db: RemoteDb, deps: ApiDeps, kind: SyncTickKind): Promise<SyncRunDto> {
  return syncNow(db, deps, kind)
}

export { CLOCK_SKEW_WARN_MS }
```

```ts
    value(db, SYNC_KEYS.lastPushAt), value(db, SYNC_KEYS.lastPullAt), value(db, SYNC_KEYS.skewMs), value(db, SYNC_KEYS.baseUrl),
  ])
  const dead = await db.select().from(s.outbox).where(eq(s.outbox.status, 'dead')).all()
  const skewMs = Number(skew ?? '0')
  const lastSyncAt = [lastPush, lastPull].filter((x): x is string => x !== null).sort().at(-1) ?? null
  return {
    registered: baseUrl !== null,
    serverBaseUrl: baseUrl,
    pendingItems: await countPendingSyncItems(db),
    deadItems: dead.length,
    lastPushAt: lastPush,
    lastPullAt: lastPull,
    lastSyncAt,
    clockSkewMs: skewMs,
    clockOff: Math.abs(skewMs) > CLOCK_SKEW_WARN_MS,
  }
}

export async function listDeadRows(db: RemoteDb): Promise<DeadRowDto[]> {
  const rows = await db.select().from(s.outbox).where(eq(s.outbox.status, 'dead')).orderBy(asc(s.outbox.createdAt)).all()
  return rows.map((r) => {
    const [reason = '', ...rest] = (r.lastError ?? '').split(': ')
    return { id: r.id, table: r.tableName, idempotencyKey: r.idempotencyKey, reason, detail: rest.join(': '), createdAt: r.createdAt, deadAt: r.deadAt }
  })
}

/** Q5-12: only the owner presses this, and only after the cause is fixed. The row goes back to pending as-is. */
export async function retryDeadRow(db: RemoteDb, deps: ApiDeps, outboxId: string): Promise<void> {
  await db.update(s.outbox).set({ status: 'pending', deadAt: null, lastError: null }).where(and(eq(s.outbox.id, outboxId), eq(s.outbox.status, 'dead')))
  void deps
}
```

```ts
import { useEffect, useRef } from 'react'
import { useApi } from './api-context'

/** spec §6.2 with D15's rule "sync เฉพาะเวลาเปิดร้าน". */
export const SYNC_INTERVALS = { pushDebounceMs: 5_000, pullMs: 300_000, flagMs: 30_000 } as const

export type SchedulerInput = { shiftOpen: boolean; pendingItems: number }

/**
 * While a shift is open: push 5 s after the outbox gains a row, pull every 5 minutes, poll the (database-free)
 * inbox flag every 30 s. With no shift open: keep pushing what is left, then stop completely so Neon stays asleep.
 */
export function useSyncScheduler({ shiftOpen, pendingItems }: SchedulerInput): void {
  const api = useApi()
  const running = useRef(false)

  const tick = async (kind: 'push' | 'pull' | 'flag'): Promise<void> => {
    if (running.current) return                    // never stack runs: the Worker queue is serial anyway
    running.current = true
    try { await api.syncTick(kind) } finally { running.current = false }
  }

  useEffect(() => {
    if (pendingItems === 0) return
    const id = setTimeout(() => { void tick('push') }, SYNC_INTERVALS.pushDebounceMs)
    return () => { clearTimeout(id) }
  }, [pendingItems])

  useEffect(() => {
    if (!shiftOpen) return
    const pull = setInterval(() => { void tick('pull') }, SYNC_INTERVALS.pullMs)
    const flag = setInterval(() => { void tick('flag') }, SYNC_INTERVALS.flagMs)
    return () => { clearInterval(pull); clearInterval(flag) }
  }, [shiftOpen])
}
```

```ts
import { eq } from 'drizzle-orm'
import type { RemoteDb } from '@dayo/db-schema/browser'
import * as s from '@dayo/db-schema/sqlite'
import { PULL_TABLES, RegisterDeviceResponse, type PullTable, type RegisterDeviceRequest } from '@dayo/contracts'
import type { ApiDeps } from '../api/deps'
import { requireDevice } from '../api/bootstrap'
import { PosError } from '../api/errors'
import type { RegisterDeviceInput, RestoreDeviceInput } from '../api/types'
import { SYNC_KEYS } from './keys'
import { pullOnce } from './pull'
import { createTransport } from './transport'

/** Reference tables a fresh tablet needs before it can sell (spec §6.4), plus the cost cache. */
const BOOTSTRAP_TABLES: readonly PullTable[] = PULL_TABLES.filter((t) => !['order', 'order_line', 'order_event', 'payment'].includes(t))

async function put(db: RemoteDb, key: string, value: string): Promise<void> {
  await db.insert(s.syncState).values({ key, value }).onConflictDoUpdate({ target: s.syncState.key, set: { value } })
}

async function alreadyRegistered(db: RemoteDb): Promise<boolean> {
  return (await db.select().from(s.syncState).where(eq(s.syncState.key, SYNC_KEYS.token)).get()) !== undefined
}

/**
 * Q5-9 · Q5-10: one-time handshake. The setup screen of plan 3 already created this device, its owners and the
 * PromptPay setting; here they travel up once (pin_hash only, never the PIN) and the tablet keeps the token.
 */
export async function registerDevice(db: RemoteDb, deps: ApiDeps, input: RegisterDeviceInput): Promise<void> {
  const device = await requireDevice(db)                       // throws NEEDS_SETUP when the tablet is blank
  if (await alreadyRegistered(db)) throw new PosError('ALREADY_REGISTERED', 'this tablet is already connected to a server')

  const users = await db.select().from(s.user).all()
  const settings = await db.select().from(s.setting).all()
  const body: RegisterDeviceRequest = {
    registrationCode: input.registrationCode,
    deviceId: device.id,
    deviceName: device.name,
    receiptPrefix: device.receiptPrefix,
    users: users.map((u) => ({ id: u.id, displayName: u.displayName, role: u.role, pinHash: u.pinHash, isActive: u.isActive, createdAt: u.createdAt })),
    settings: settings.map((st) => ({ key: st.key, valueJson: st.valueJson, effectiveFrom: st.effectiveFrom })),
  }

  const res = await deps.fetch(`${input.baseUrl}/auth/device/register`, {
    method: 'POST', headers: { 'content-type': 'application/json' }, body: JSON.stringify(body),
  })
  if (res.status === 401 || res.status === 403) throw new PosError('SYNC_UNAUTHORIZED', 'the registration code was refused')
  if (!res.ok) throw new PosError('SYNC_OFFLINE', `${res.status}`)
  const answer = RegisterDeviceResponse.parse(await res.json())

  await put(db, SYNC_KEYS.baseUrl, input.baseUrl)
  await put(db, SYNC_KEYS.token, answer.token)

  const transport = await createTransport(db, deps)
  if (transport !== null) await pullOnce(db, transport, BOOTSTRAP_TABLES)
}

/**
 * spec §6.4: a replacement tablet. Setup gave it a new id and a new receipt prefix, so registering is the same
 * call; what differs is that it must also pull recent bills for the history screen (Q5-11 — 30 days, no older).
 */
export async function restoreDevice(db: RemoteDb, deps: ApiDeps, input: RestoreDeviceInput): Promise<void> {
  await registerDevice(db, deps, input)
  const transport = await createTransport(db, deps)
  if (transport === null) throw new PosError('NOT_REGISTERED', 'registration did not complete')
  await pullOnce(db, transport, ['order', 'order_line', 'order_event', 'payment'])
}
```

`scripts/backup/verify-dump.sh` (ไม่ผ่าน = งานล้ม = เจ้าของได้เมลจาก GitHub):

```bash
#!/usr/bin/env bash
# A dump that cannot be listed is not a backup. Fails the job loudly instead of uploading a broken file.
set -euo pipefail
file="$1"
size=$(wc -c < "$file")
if [ "$size" -lt 50000 ]; then echo "dump is only $size bytes — refusing" >&2; exit 1; fi
tables=$(pg_restore --list "$file" | grep -c 'TABLE DATA' || true)
echo "dump: $size bytes · $tables tables with data"
if [ "$tables" -lt 20 ]; then echo "only $tables tables with data — refusing" >&2; exit 1; fi
```

`scripts/backup/upload-to-r2.sh`:

```bash
#!/usr/bin/env bash
# Keeps 30 daily copies plus one copy per month for 12 months (spec §10 · Q5-14).
set -euo pipefail
file="$1"; day="$2"
endpoint="https://${R2_ACCOUNT_ID}.r2.cloudflarestorage.com"
aws="aws --endpoint-url $endpoint --region auto"

$aws s3 cp "$file" "s3://${R2_BUCKET}/dayo-pos/daily/${day}.dump"
if [ "${day:8:2}" = "01" ]; then
  $aws s3 cp "$file" "s3://${R2_BUCKET}/dayo-pos/monthly/${day:0:7}.dump"
fi

cutoff=$(date -u -d '30 days ago' +%F)
$aws s3 ls "s3://${R2_BUCKET}/dayo-pos/daily/" | awk '{print $4}' | while read -r key; do
  [ -z "$key" ] && continue
  if [[ "${key%.dump}" < "$cutoff" ]]; then $aws s3 rm "s3://${R2_BUCKET}/dayo-pos/daily/${key}"; fi
done

cutoff_month=$(date -u -d '12 months ago' +%Y-%m)
$aws s3 ls "s3://${R2_BUCKET}/dayo-pos/monthly/" | awk '{print $4}' | while read -r key; do
  [ -z "$key" ] && continue
  if [[ "${key%.dump}" < "$cutoff_month" ]]; then $aws s3 rm "s3://${R2_BUCKET}/dayo-pos/monthly/${key}"; fi
done
```
